local Players = game:GetService("Players")
local Teams = game:GetService("Teams")
local Workspace = game:GetService("Workspace")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

-- Create or find RemoteEvents
local hudEvent = ReplicatedStorage:FindFirstChild("HUDUpdate")
if not hudEvent then
	hudEvent = Instance.new("RemoteEvent")
	hudEvent.Name = "HUDUpdate"
	hudEvent.Parent = ReplicatedStorage
end

local scoreEvent = ReplicatedStorage:FindFirstChild("ScoreUpdate")
if not scoreEvent then
	scoreEvent = Instance.new("RemoteEvent")
	scoreEvent.Name = "ScoreUpdate"
	scoreEvent.Parent = ReplicatedStorage
end

-- NEW: RemoteEvent for Snowball Hits
local hitEvent = ReplicatedStorage:FindFirstChild("SnowballHit")
if not hitEvent then
	hitEvent = Instance.new("RemoteEvent")
	hitEvent.Name = "SnowballHit"
	hitEvent.Parent = ReplicatedStorage
end

local teamScores = {["Red Team"] = 0, ["Blue Team"] = 0, ["Green Team"] = 0}
local playerKills = {}

-- Team setup
local function getTeamByExactName(name, color)
	local t = Teams:FindFirstChild(name)
	if not t then
		t = Instance.new("Team")
		t.Name = name
		t.TeamColor = color
		t.Parent = Teams
	end
	return t
end

local redTeam = getTeamByExactName("Red Team", BrickColor.new("Bright red"))
local blueTeam = getTeamByExactName("Blue Team", BrickColor.new("Bright blue"))
local greenTeam = getTeamByExactName("Green Team", BrickColor.new("Bright green"))
local teamList = {redTeam, blueTeam, greenTeam}

local function assignTeam(player)
	local targetTeam = teamList[1]
	local minCount = #targetTeam:GetPlayers()
	for _, team in ipairs(teamList) do
		local count = #team:GetPlayers()
		if count < minCount then
			minCount = count
			targetTeam = team
		end
	end
	player.Team = targetTeam
end

-- Centralized Kill / Point Scorer
local function recordKill(killerPlayer, victimPlayer)
	if not killerPlayer or not victimPlayer then return end
	if killerPlayer == victimPlayer then return end -- No self-harm points

	if not killerPlayer.Team or killerPlayer.Team == Teams.Neutral then
		assignTeam(killerPlayer)
	end

	if victimPlayer.Team and killerPlayer.Team == victimPlayer.Team then
		return -- Ignore friendly fire
	end

	local tName = killerPlayer.Team.Name
	if teamScores[tName] then
		teamScores[tName] = teamScores[tName] + 1
	else
		teamScores[tName] = 1
	end

	playerKills[killerPlayer] = (playerKills[killerPlayer] or 0) + 1
	scoreEvent:FireAllClients(teamScores, playerKills)
	print("[RoundManager] POINT SCORED! " .. tName .. " now has " .. teamScores[tName] .. " points.")
end

-- Listen for hits sent from the Snowball tool
hitEvent.OnServerEvent:Connect(function(killerPlayer, victimPlayer)
	if roundState == "ACTIVE" then
		recordKill(killerPlayer, victimPlayer)
	end
end)

-- Keep global hook just in case other scripts use it
_G.RecordSnowballKill = recordKill

local function broadcastHUD(statusText, winnerText)
	hudEvent:FireAllClients(statusText, winnerText, teamScores)
end

local roundState = "WAITING"
local waitingDuration = 15
local roundDuration = 30

_G.SetRoundTime = function(newTime)
	roundDuration = newTime
end

_G.SetIntermissionTime = function(newTime)
	waitingDuration = newTime
end

local function sendEveryoneToLobby()
	local lobbySpawn = Workspace:FindFirstChild("LobbySpawn")
	for _, player in ipairs(Players:GetPlayers()) do
		if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
			if lobbySpawn then
				player.Character.HumanoidRootPart.CFrame = lobbySpawn.CFrame + Vector3.new(0, 3, 0)
			end
		end
	end
end

Players.PlayerAdded:Connect(function(player)
	assignTeam(player)
	playerKills[player] = 0

	player.CharacterAdded:Connect(function(character)
		local hrp = character:WaitForChild("HumanoidRootPart", 5)
		task.wait(0.2)

		if hrp then
			if roundState == "ACTIVE" then
				local teamSpawn = Workspace:FindFirstChild(player.Team.Name .. "Spawn")
				if teamSpawn then
					hrp.CFrame = teamSpawn.CFrame + Vector3.new(0, 3, 0)
				end
			else
				local lobbySpawn = Workspace:FindFirstChild("LobbySpawn")
				if lobbySpawn then
					hrp.CFrame = lobbySpawn.CFrame + Vector3.new(0, 3, 0)
				end
			end
		end
	end)
end)

Players.PlayerRemoving:Connect(function(player)
	playerKills[player] = nil
end)

-- Main Game Loop
task.spawn(function()
	while true do
		-- 1. WAITING / INTERMISSION
		roundState = "WAITING"
		teamScores = {["Red Team"] = 0, ["Blue Team"] = 0, ["Green Team"] = 0}
		scoreEvent:FireAllClients(teamScores, playerKills)

		sendEveryoneToLobby()
		local waitTimer = waitingDuration
		while waitTimer > 0 and roundState == "WAITING" do
			broadcastHUD("Waiting for players: " .. waitTimer .. "s", "")
			task.wait(1)
			waitTimer = waitTimer - 1
		end

		-- 2. PREP PHASE
		roundState = "PREP"
		sendEveryoneToLobby()
		for i = 10, 1, -1 do
			broadcastHUD("Get your snowballs! " .. i .. "s", "")
			task.wait(1)
		end

		-- 3. ACTIVE ROUND
		roundState = "ACTIVE"
		for _, player in ipairs(Players:GetPlayers()) do
			if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
				local charHrp = player.Character.HumanoidRootPart
				local teamSpawn = Workspace:FindFirstChild(player.Team.Name .. "Spawn")
				if teamSpawn then
					charHrp.CFrame = teamSpawn.CFrame + Vector3.new(0, 3, 0)
				end
			end
		end

		local timeLeft = roundDuration
		while timeLeft > 0 and roundState == "ACTIVE" do
			broadcastHUD("Round Time: " .. timeLeft .. "s", "")
			task.wait(1)
			timeLeft = timeLeft - 1
		end

		-- 4. ROUND END
		roundState = "END"
		broadcastHUD("Round Over! Calculating...", "")

		local winningTeam = "Tie"
		local maxScore = -1
		for tName, score in pairs(teamScores) do
			if score > maxScore then
				maxScore = score
				winningTeam = tName
			elseif score == maxScore and maxScore > 0 then
				winningTeam = "Tie"
			end
		end

		local winText = winningTeam == "Tie" and "It's a Tie!" or (winningTeam .. " Wins!")
		broadcastHUD("Round Ended", winText)
		task.wait(3)
		sendEveryoneToLobby()
		task.wait(4)
	end
end)
